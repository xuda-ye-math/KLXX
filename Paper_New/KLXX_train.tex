\begin{tikzpicture}
    % math glue is fixed, so relations in text-width nodes are not stretched
    \medmuskip=4mu \thickmuskip=5mu

    % --- prepared once, before the loop ---
    \node[topbar] (inputs) at (0,5.55) {%
        \textbf{Prepared before training}\\[0.5mm]
        {\small source samples $\mathcal Y_0\sim\pi_0$ of size $N$ $\cdot$ QT samples $\hat{\mathcal Y}=\mathrm{QT}(\mathcal Y_0,U)$ of size $N$}
    };

    % --- first row: the three sample sets ---
    \node[stepbox, draw=proposal, fill=proposalfill] (s1) at (-6.0,3.3) {%
        \textbf{(i) Pushforward $\mathcal X_{\nu}$}\\[0.8mm]
        {\small draw $\mathcal X_0\subset\mathcal Y_0$ of size $B$}\\[-0.2mm]
        {\small compute $\mathcal X_{\nu}=G^{-1}(\mathcal X_0)$}
    };

    \node[stepbox, draw=target, fill=targetfill] (s2) at (0,3.3) {%
        \textbf{(ii) Target surrogate $\mathcal X_{\pi}$}\\[0.8mm]
        {\small apply AIS on $\mathcal X_0$ to obtain $\mathcal X_{\pi}$}
    };

    \node[stepbox, draw=coverage, fill=coveragefill] (s3) at (6.0,3.3) {%
        \textbf{(iii) Mixture $\mathcal X_{\xi}$}\\[0.8mm]
        {\small draw $\mathcal X_{\hat\pi}\subset\hat{\mathcal Y}$ from QT}\\[-0.2mm]
        {\small mixed with $\mathcal X_{\nu}$ at $(\alpha,\beta)$}
    };

    \draw[flow] (s1.north |- inputs.south) -- (s1.north);
    \draw[flow] (s3.north |- inputs.south) -- (s3.north);

    \draw[flow] (s1.east) -- (s2.west);
    \draw[flow] (s2.east) -- (s3.west);

    % --- second row: the gradient step and the loss it evaluates ---
    \node[stepbox] (s4) at (-6.0,0.1) {%
        \textbf{(iv) Gradient step}\\[0.8mm]
        {\small evaluate $\widehat{\mathcal L}[G]$ and update $G$}\\[-0.2mm]
        {\small only $z(y)$ differentiated}
    };

    \node[eqpanel] (loss) at (2.3,0.1) {%
        $\displaystyle
        \begin{aligned}
        \widehat{\mathcal L}[G] = {}
          & \underbrace{\widehat{\mathbb E}_{y\in\mathcal X_\pi}\bigl[z(y)\bigr]}_{\text{forward KL}}
          + \lambda\underbrace{\widehat{\mathbb E}_{y,y'\in\mathcal X_\pi}
            \bigl[|z(y)-z(y')|\bigr]}_{\text{target variation}}\\[1.4mm]
          & + \vartheta\underbrace{\widehat{\mathbb E}_{y,y'\in\mathcal X_{\xi}}
            \bigl[|z(y)-z(y')|\bigr]}_{\text{mixture variation}}
          \quad \xi = \alpha\hat\pi+\beta\bar\nu
        \end{aligned}$
    };

    % --- (iii) feeds the loss, the loss feeds the gradient step ---
    \draw[flow] (s3.south) -- (s3.south |- loss.north);
    \draw[flow] (loss.west) -- (s4.east);

    % --- the loop: back to the first step for the next batch ---
    \draw[flow] (s4.north) -- node[left=1mm, note] {next batch} (s1.south);

\end{tikzpicture}
